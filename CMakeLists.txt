cmake_minimum_required(VERSION 3.20)
project(SANGUE VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_compile_options(/W4 /utf-8)
endif()
enable_testing()
add_executable(model_tests tests/model_test.cpp)
target_include_directories(model_tests PRIVATE src)
add_test(NAME model_contracts COMMAND model_tests)
if(WIN32)
  enable_language(RC)
  configure_file(src/app.rc.in app.rc @ONLY)
  add_executable(SANGUE WIN32 src/main.cpp "${CMAKE_CURRENT_BINARY_DIR}/app.rc")
  target_compile_definitions(SANGUE PRIVATE UNICODE _UNICODE NOMINMAX _WIN32_WINNT=0x0A00 SANGUE_VERSION="${PROJECT_VERSION}")
  target_link_libraries(SANGUE PRIVATE gdiplus gdi32 user32 dwmapi bcrypt ole32 shell32)
  if(MSVC)
    target_link_options(SANGUE PRIVATE /MANIFEST:NO /OPT:REF /OPT:ICF)
  endif()
  add_executable(runtime_tests tests/runtime_test.cpp)
  target_include_directories(runtime_tests PRIVATE src)
  target_compile_definitions(runtime_tests PRIVATE UNICODE _UNICODE NOMINMAX _WIN32_WINNT=0x0A00)
  target_link_libraries(runtime_tests PRIVATE bcrypt user32)
  add_test(NAME windows_runtime COMMAND runtime_tests)
  add_test(NAME native_panel COMMAND SANGUE --capture "${CMAKE_CURRENT_BINARY_DIR}/panel.png")
  set_tests_properties(native_panel PROPERTIES TIMEOUT 30)
endif()
